MODULE backend.services.checkpoint_runtime
DOC: One server-owned clock and one decision per human checkpoint round.

No model calls. Runtime identities deliberately expire on process restart: an
old browser must never approve a recovered/new round with its previous token.
CALLABLES:
_finish(session: 'Checkpoint', response: 'dict') -> 'None' :: 
close(session: 'Checkpoint') -> 'None' :: 
current(workflow_id: 'str') -> 'dict | None' :: 
require_live(workflow_id: 'str', checkpoint_id: 'str | None') -> 'Checkpoint' :: 
resolve(workflow_id: 'str', checkpoint_id: 'str | None', response: 'dict') -> 'dict' :: 
start(workflow_id: 'str', payload: 'dict', timeout: 'float' = 1800, on_timeout: 'str' = 'approve') -> 'Checkpoint' :: 
wait(session: 'Checkpoint', *, release: 'bool' = True) -> 'dict' ::
