==================== workflows
CREATE TABLE workflows (
    id TEXT PRIMARY KEY,
    template TEXT NOT NULL,          -- idea_discovery | experiment_bridge | auto_review | paper_writing | full_pipeline | thesis_proposal | literature_review | course_paper | course_report | humanities_paper | comp_* ...
    title TEXT NOT NULL,             -- 用户输入的研究方向
    params TEXT DEFAULT '{}',        -- JSON: AUTO_PROCEED, HUMAN_CHECKPOINT 等
    status TEXT DEFAULT 'pending',   -- pending | running | paused | completed | failed
    current_step TEXT,               -- 当前执行的 skill 名称
    workspace_dir TEXT,              -- 工作区目录路径
    enable_checkpoints INTEGER DEFAULT 0,  -- 0=关闭检查点(自动连续执行) 1=开启检查点(步骤完成后暂停确认)
    -- 自动恢复护栏必须跨后端重启持久化，不能只放在 Python 内存字典里。
    -- 人工 start/resume/restart/rerun 会清零；心跳/启动恢复只递增。
    auto_resume_count INTEGER NOT NULL DEFAULT 0,
    auto_resume_last_at TIMESTAMP,
    -- 对话模式"区间运行"的范围（JSON：起止步骤 id、本次要求）。不放 params：params 会写进
    -- CLAUDE.md、被多处读改写；范围要跨重启/心跳复活持续生效，跑到终点即清（见 load_run_scope）。
    run_scope TEXT,
    -- 用户自己起的项目名（10-20 内测反馈“希望能给任务增加自命名环节”）：只管左栏、首页、顶栏怎么显示，
    -- title 仍是研究方向 / 题目（写进 CLAUDE.md、每一步按它做）。空 = 没另起名，照旧显示题目。
    custom_name TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)

==================== workflow_steps
CREATE TABLE workflow_steps (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    workflow_id TEXT NOT NULL REFERENCES workflows(id),
    skill_name TEXT NOT NULL,
    display_name TEXT NOT NULL,
    step_order INTEGER NOT NULL,
    status TEXT DEFAULT 'pending',   -- pending | running | waiting_checkpoint | completed | failed | skipped
    has_checkpoint INTEGER DEFAULT 0,
    checkpoint_type TEXT,            -- idea_select | approve | feedback
    output_files TEXT DEFAULT '[]',  -- JSON array
    started_at TIMESTAMP,
    completed_at TIMESTAMP,
    error_message TEXT,
    model_used TEXT,                 -- 该步真跑时实际生效的 model_id 快照（预设后来被删/改也不影响历史回看）
    resume_session_id TEXT,          -- 仅本步骤的本地 Claude 会话；服务端请求/计费从不复用
    resume_error_code TEXT,          -- 允许前端展示显式恢复选择的类型化原因
    resume_strategy TEXT,            -- NULL | same_session | fresh_context
    vision_review TEXT,              -- JSON：本步看图质检结论汇总（services/step_review_report.py），只做展示
    precompile_report TEXT,          -- JSON：本步编译预处理改了哪些文件、清单与快照位置，只做展示
    submission_check TEXT,           -- JSON：论文编译步的投稿前检查汇总（services/submission_review.py），只做展示
    writing_check TEXT,              -- JSON：本步的写作检查汇总（规划步的论证主线核对，services/writing_review.py），只做展示
    handoff_reason TEXT,             -- 上次失败/检查暂停/中断的原因，下次执行本步时交给模型；本步完成即清空
    main_output_receipt TEXT,        -- JSON：主调用成功的回执；继续时有效就不再付费重做整步
    paid_continues INTEGER NOT NULL DEFAULT 0  -- 本步没完成时连续付费"继续"的次数，到上限就停下给出路
)

==================== checkpoints
CREATE TABLE checkpoints (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    workflow_id TEXT NOT NULL REFERENCES workflows(id),
    step_name TEXT NOT NULL,
    checkpoint_type TEXT NOT NULL,
    data TEXT DEFAULT '{}',          -- JSON: 展示给用户的数据
    response TEXT,                   -- JSON: 用户的回复
    status TEXT DEFAULT 'pending',   -- pending | resolved
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    resolved_at TIMESTAMP
)

==================== workflow_logs
CREATE TABLE workflow_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    workflow_id TEXT NOT NULL REFERENCES workflows(id),
    step_name TEXT,
    level TEXT DEFAULT 'info',       -- info | warn | error | progress
    message TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)
